export function pigeonWing(ph, amt) {
  const e = Math.cos(TAU * ph) * amt, eh = Math.cos(TAU * (ph - 0.08)) * amt, fold = Math.max(0, -Math.sin(TAU * (ph - 0.04))) * amt;
  return { e, fold, ea: 0.14 + 0.62 * e, eh: 0.04 + 0.92 * eh - 0.3 * fold };
}
